const { checkEnvironment } = await import(pathToFileURL(`${process.cwd()}/scripts/check-merge.mjs`));
const files = ['api/http-session-start.concurrent', 'api/http-session-start',
  'api/http-session-start.steward-probe', 'api/orchestrator-assignment',
  'runtime/session-manager', 'services/dispatch-claim',
  'services/task-assignment-service', 'services/task-completion-protocol'];
let failed = false;
for (const file of files) {
  const result = spawnSync('bun', ['test', `packages/smithy/src/${file}.bun.test.ts`],
    { env: checkEnvironment(), stdio: 'inherit' });
  if (result.status !== 0) failed = true;
}
process.exitCode = failed ? 1 : 0;
